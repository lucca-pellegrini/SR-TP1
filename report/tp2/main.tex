\newcommand{\pentrada}[3]{\draw[->,#3] ($(bloco.west)+(-\fio,#1)$) -- ($(bloco.west)+(0,#1)$);
  \node[port, anchor=east] at ($(bloco.west)+(-\fio-0.1,#1)$) {#2};}
\newcommand{\psaida}[3]{\draw[->,#3] ($(bloco.east)+(0,#1)$) -- ($(bloco.east)+(\fio,#1)$);
